Land the cashout endpoint: a synchronous 200 OK response carrying the payout. Cashout is a 1-step saga (REQ-SAGA-04) — Game atomically transitions Bet → CASHED_OUT and writes the `wallet.credit` outbox row in the same TX, then returns to the client immediately.        - Add the method:
         ```ts
         @Post('cashout')
         @HttpCode(200)
         async cashout(@Req() req: AuthenticatedRequest): Promise<CashoutResponseDto> {
           const acceptedAt = new Date();   // FIRST line, BEFORE any await — Pitfall C2 prevention
           const playerId = PlayerId(req.user!.playerId);
           let multiplier: Multiplier;
           try {
             multiplier = this.roundLoop.getMultiplierAt(acceptedAt);
           } catch {
             // No RUNNING round in memory — surface as 409 ROUND_NOT_RUNNING
             throw new ConflictException({ code: 'ROUND_NOT_RUNNING', phase: 'UNKNOWN' });
           }
           try {
             const result = await this.cashOut.execute({ playerId, multiplier, acceptedAt });
             return { multiplier: result.multiplier.toNumber(), payoutCents: result.payout.toSnapshot() };
           } catch (err) {
             this.translateCashoutError(err);
           }
         }

         private translateCashoutError(err: unknown): never {
           if (err instanceof RoundNotRunningError) throw new ConflictException({ code: 'ROUND_NOT_RUNNING', phase: err.actual });
           if (err instanceof NoActiveBetError) throw new ConflictException({ code: 'NO_ACTIVE_BET' });
           if (err instanceof BetNotCashableError) throw new ConflictException({ code: 'BET_NOT_CASHABLE', status: err.status });
           throw err;
         }
         ```
