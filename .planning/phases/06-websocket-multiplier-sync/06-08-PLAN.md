Boot the app once per `fc.assert` invocation is too costly — instead, boot once in `beforeAll` and rely on round-loop natural cadence; each `asyncProperty` case picks `dtMs` and pegs its cashout to the NEXT RUNNING round.

    Outline:
    ```
    beforeAll: applyTestEnv with BETTING_WINDOW_MS=2000, COOLDOWN_MS=500, INSTANT_CRASH_BUCKET=1000000; bootApp(); listen on ephemeral port; provision wallet; await first round
    afterAll: app.close()
    beforeEach: truncate bets + bet_saga_state; ensure wallet has >= bet amount

    test("cashout ±50ms of crash never double-pays and never pays post-crash", async () => {
      await fc.assert(fc.asyncProperty(fc.integer({ min: -50, max: 50 }), async (dtMs) => {
        await waitForPhase("BETTING");
        const bet = await placeBetViaKong({ amountCents: "1000" }, token);
        await waitForBetActive(bet.id);
        await waitForPhase("RUNNING");
        const round = await getCurrentRound();
        const targetAt = computeCrashAtMs(round) + dtMs;
        await sleepUntil(targetAt);
        const res = await cashoutViaKong(token);
        const cashoutCount = await em.count(BetEntity, { id: bet.id, status: "CASHED_OUT" });
        expect(cashoutCount).toBeLessThanOrEqual(1);
        if (res.status === 200) {
          const dbBet = await em.findOne(BetEntity, { id: bet.id });
          const dbRound = await em.findOne(RoundEntity, { id: round.id });
          expect(dbBet?.cashedOutAt.getTime()).toBeLessThan(dbRound?.crashedAt.getTime() ?? Infinity);
        } else {
          expect(res.status).toBe(409);
          expect(["ROUND_NOT_RUNNING", "BET_NOT_CASHABLE", "NO_ACTIVE_BET"]).toContain(res.body.code);
        }
      }), { numRuns: 50 });
    });
    ```
